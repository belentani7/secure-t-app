```typescript
// src/lib/db/schema.ts


import { pgTable, serial, text, integer, boolean, timestamp, jsonb } from 'drizzle-orm/pg-core';


// TABLA USUARIOS
export const users = pgTable('users', {
  id: serial('id').primaryKey(),
  email: text('email').unique().notNull(),
  name: text('name').notNull(),
  role: text('role').default('student'), // student, instructor, admin
  passwordHash: text('password_hash').notNull(),
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow(),
});


// TABLA CURRÍCULO (AÑOS ACADÉMICOS)
export const academicYears = pgTable('academic_years', {
  id: serial('id').primaryKey(),
  year: integer('year').notNull(), // 1, 2, 3, 4
  title: text('title').notNull(),
  description: text('description'),
  createdAt: timestamp('created_at').defaultNow(),
});


// TABLA CURSOS
export const courses = pgTable('courses', {
  id: serial('id').primaryKey(),
  academicYearId: integer('academic_year_id').references(() => academicYears.id),
  code: text('code').unique().notNull(), // ej: "LINUX-101"
  title: text('title').notNull(),
  description: text('description'),
  category: text('category'), // linux, networking, python, cybersecurity, websec, detection, ir
  credits: integer('credits').default(3),
  createdAt: timestamp('created_at').defaultNow(),
});


// TABLA LECCIONES
export const lessons = pgTable('lessons', {
  id: serial('id').primaryKey(),
  courseId: integer('course_id').references(() => courses.id),
  title: text('title').notNull(),
  content: text('content').notNull(),
  type: text('type'), // theory, practical, lab, assessment
  order: integer('order').default(0),
  learningObjectives: jsonb('learning_objectives').default([]),
  competencies: jsonb('competencies').default([]),
  createdAt: timestamp('created_at').defaultNow(),
});


// TABLA EVIDENCIAS
export const evidences = pgTable('evidences', {
  id: serial('id').primaryKey(),
  userId: integer('user_id').references(() => users.id),
  lessonId: integer('lesson_id').references(() => lessons.id),
  content: text('content'),
  fileUrl: text('file_url'),
  status: text('status').default('pending'), // pending, approved, rejected
  submittedAt: timestamp('submitted_at').defaultNow(),
  reviewedAt: timestamp('reviewed_at'),
});


// TABLA NOTIFICACIONES
export const notifications = pgTable('notifications', {
  id: serial('id').primaryKey(),
  userId: integer('user_id').references(() => users.id),
  title: text('title').notNull(),
  message: text('message').notNull(),
  type: text('type'), // task, system, achievement
  read: boolean('read').default(false),
  data: jsonb('data').default({}),
  createdAt: timestamp('created_at').defaultNow(),
});


// TABLA AUDITORÍA
export const auditLogs = pgTable('audit_logs', {
  id: serial('id').primaryKey(),
  userId: integer('user_id').references(() => users.id),
  action: text('action').notNull(),
  resource: text('resource'),
  details: jsonb('details').default({}),
  ip: text('ip'),
  userAgent: text('user_agent'),
  createdAt: timestamp('created_at').defaultNow(),
});


// TABLA CONFIGURACIÓN IA
```

### **A. API del Currículo**


```typescript
// src/app/api/curriculum/route.ts


import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { academicYears, courses, lessons } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { logger } from '@/lib/logger';
import { authenticate } from '@/lib/auth';


export async function GET(request: Request) {
  try {
    // Autenticación
    const user = await authenticate(request);
    if (!user) {
      return NextResponse.json({ error: 'No autorizado' }, { status: 401 });
    }


    // Obtener todos los años académicos con sus cursos
    const years = await db.select().from(academicYears);
    
    const curriculum = await Promise.all(years.map(async (year) => {
      const yearCourses = await db.select()
        .from(courses)
        .where(eq(courses.academicYearId, year.id));
      
      const coursesWithLessons = await Promise.all(yearCourses.map(async (course) => {
        const courseLessons = await db.select()
          .from(lessons)
          .where(eq(lessons.courseId, course.id))
          .orderBy(lessons.order);
        
        return {
          ...course,
          lessons: courseLessons
        };
      }));
      
      return {
        ...year,
        courses: coursesWithLessons
      };
    }));


    // Log de auditoría
    await logger.audit({
      userId: user.id,
      action: 'VIEW_CURRICULUM',
      resource: 'curriculum',
      details: { yearCount: years.length }
    });


    return NextResponse.json(curriculum);
  } catch (error) {
    console.error('Error fetching curriculum:', error);
    return NextResponse.json(
      { error: 'Error al obtener el currículo' },
      { status: 500 }
    );
  }
}
```


### **B. Sistema de Notificaciones**


```typescript
// src/app/components/Notifications.tsx


import { useEffect, useState } from 'react';
import { useNotifications } from '@/hooks/useNotifications';
import { useAudio } from '@/hooks/useAudio';


interface Notification {
  id: number;
  title: string;
  message: string;
  type: string;
  read: boolean;
  data?: any;
  createdAt: string;
}


export function Notifications() {
  const [notifications, setNotifications] = useState<Notification[]>([]);
  const [permission, setPermission] = useState<NotificationPermission>('default');
  const { playSound } = useAudio();


  // Solicitar permiso de notificaciones
  useEffect(() => {
    if ('Notification' in window) {
      setPermission(Notification.permission);
    }
  }, []);


  const requestPermission = async () => {
    if ('Notification' in window) {
      const perm = await Notification.requestPermission();
      setPermission(perm);
      if (perm === 'granted') {
        // Registrar Service Worker
        if ('serviceWorker' in navigator) {
          navigator.serviceWorker.register('/sw.js');
        }
      }
    }
  };


  // Mostrar notificación
  const showNotification = (notification: Notification) => {
    // Sonido
    playSound('/notification.mp3');


    // Notificación del navegador
    if ('Notification' in window && Notification.permission === 'granted') {
      new Notification(notification.title, {
        body: notification.message,
        icon: '/icon-192.png',
        badge: '/badge.png',
        vibrate: [200, 100, 200],
        data: notification.data,
      });
    }


    // Agregar al estado
    setNotifications(prev => [notification, ...prev]);


    // Enviar a la API para guardar
    fetch('/api/notifications', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(notification),
    });
  };


  const markAsRead = async (id: number) => {
    await fetch(`/api/notifications/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ read: true }),
    });


    setNotifications(prev =>
      prev.map(n => n.id === id ? { ...n, read: true } : n)
    );
  };


  return (
    <div className="notifications-container">
      {permission !== 'granted' && (
        <button onClick={requestPermission}>
          Activar Notificaciones
        </button>
      )}
      
      <div className="notifications-list">
        {notifications.map((notif) => (
          <div key={notif.id} className={`notification ${notif.read ? 'read' : 'unread'}`}>
            <h3>{notif.title}</h3>
            <p>{notif.message}</p>
            <span className="timestamp">
              {new Date(notif.createdAt).toLocaleString()}
            </span>
            {!notif.read && (
              <button onClick={() => markAsRead(notif.id)}>
                Marcar como leída
              </button>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
```


### **C. Service Worker**


```javascript
// public/sw.js


self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open('v1').then((cache) => {
      return cache.addAll([
        '/',
        '/manifest.json',
        '/icon-192.png',
        '/notification.mp3'
      ]);
    })
  );
  self.skipWaiting();
});


self.addEventListener('activate', (event) => {
  event.waitUntil(clients.claim());
});


self.addEventListener('push', (event) => {
  const data = event.data?.json() || {};
  
  const options = {
    body: data.body || 'Tarea completada',
    icon: '/icon-192.png',
    badge: '/badge.png',
    vibrate: [200, 100, 200],
    data: data.data || {},
    actions: [
      { action: 'view', title: 'Ver' },
      { action: 'dismiss', title: 'Cerrar' }
    ],
    tag: data.id || 'notification',
    renotify: true,
  };


  event.waitUntil(
    self.registration.showNotification(
      data.title || 'Secure T',
      options
    )
  );
});


// Manejar clicks en notificaciones
self.addEventListener('notificationclick', (event) => {
  event.notification.close();
  
  if (event.action === 'view') {
    const url = event.notification.data.url || '/';
    event.waitUntil(
      clients.openWindow(url)
    );
  }
});
```

```typescript
    'manage_roles',
    'view_audit_logs',
    'manage_ai_config',
    'view_notifications',
    'send_notifications',
  ],
};


export function hasPermission(userRole: UserRole, permission: Permission): boolean {
  return rolePermissions[userRole]?.includes(permission) || false;
}


// Middleware de autorización
export async function authorize(user: any, permission: Permission): Promise<boolean> {
  if (!user) return false;
  return hasPermission(user.role as UserRole, permission);
}
```

### **E. Sistema de Auditoría**


```typescript
// src/lib/logger.ts


import { db } from './db';
import { auditLogs } from './db/schema';


interface AuditLogData {
  userId: number;
  action: string;
  resource?: string;
  details?: any;
  ip?: string;
  userAgent?: string;
}


export const logger = {
  async audit(data: AuditLogData) {
    try {
      await db.insert(auditLogs).values({
        userId: data.userId,
        action: data.action,
        resource: data.resource,
        details: data.details || {},
        ip: data.ip,
        userAgent: data.userAgent,
        createdAt: new Date(),
      });
    } catch (error) {
      console.error('Error writing audit log:', error);
    }
  },


  async getAuditLogs(userId?: number, limit = 100) {
    try {
      let query = db.select().from(auditLogs);
      if (userId) {
        query = query.where({ userId });
      }
      query = query.limit(limit).orderBy(auditLogs.createdAt, 'desc');
      return await query;
    } catch (error) {
      console.error('Error fetching audit logs:', error);
      return [];
    }
  }
};
```


### **F. Gobernanza IA**


```typescript
// src/lib/governance.ts


import { db } from './db';
import { aiGovernance } from './db/schema';


interface AIRequest {
  userId: number;
  modelName: string;
  query: string;
  context: any;
}


export class AIGovernance {
  // Validar si una solicitud está permitida
  async validateRequest(request: AIRequest): Promise<{ allowed: boolean; reason?: string }> {
    const config = await this.getConfig(request.modelName);
    
    if (!config) {
      return { allowed: false, reason: 'Modelo no configurado' };
    }


    // Verificar restricciones
    for (const restriction of config.restrictions || []) {
      if (this.matchesRestriction(request, restriction)) {
        return { allowed: false, reason: `Restricción: ${restriction.reason}` };
      }
    }


    // Verificar límites de uso
    const usage = await this.getUsageStats(request.userId, request.modelName);
    if (usage >= (config.maxDailyRequests || 100)) {
      return { allowed: false, reason: 'Límite diario de solicitudes excedido' };
    }


    return { allowed: true };
  }


  async getConfig(modelName: string) {
    const configs = await db.select()
      .from(aiGovernance)
      .where({ modelName })
      .orderBy('version', 'desc')
      .limit(1);
    
    return configs[0];
  }


  async getUsageStats(userId: number, modelName: string) {
    // Implementar conteo de uso
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    
    // Obtener solicitudes del día
    const count = await db
      .select({ count: sql`count(*)` })
      .from(aiAuditLogs)
      .where({
        userId,
        modelName,
        createdAt: sql`>= ${today}`
      });
    
    return count[0]?.count || 0;
  }


  // Registrar uso de IA para auditoría
  async logAIUse(data: {
    userId: number;
    modelName: string;
    query: string;
    response: string;
    duration: number;
    cost: number;
  }) {
    await db.insert(aiAuditLogs).values({
      ...data,
      timestamp: new Date(),
    });


    // Notificar si se excede el umbral
    if (data.cost > 100) {
      await this.sendAlert('AI Cost Alert', `Alto costo: ${data.cost}`);
    }
  }


  matchesRestriction(request: AIRequest, restriction: any): boolean {
    // Implementar lógica de matching
    if (restriction.type === 'content') {
      return request.query.includes(restriction.pattern);
    }
    return false;
  }


  async sendAlert(title: string, message: string) {
    // Enviar notificación administrativa
    const admins = await db.select()
      .from(users)
      .where({ role: 'admin' });
    
    for (const admin of admins) {
      await this.sendNotification(admin.id, title, message);
    }
  }
}
```

### **A. API del Currículo**


```typescript
// src/app/api/curriculum/route.ts


import { NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { academicYears, courses, lessons } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { logger } from '@/lib/logger';
import { authenticate } from '@/lib/auth';


export async function GET(request: Request) {
  try {
    // Autenticación
    const user = await authenticate(request);
    if (!user) {
      return NextResponse.json({ error: 'No autorizado' }, { status: 401 });
    }


    // Obtener todos los años académicos con sus cursos
    const years = await db.select().from(academicYears);
    
    const curriculum = await Promise.all(years.map(async (year) => {
      const yearCourses = await db.select()
        .from(courses)
        .where(eq(courses.academicYearId, year.id));
      
      const coursesWithLessons = await Promise.all(yearCourses.map(async (course) => {
        const courseLessons = await db.select()
          .from(lessons)
          .where(eq(lessons.courseId, course.id))
          .orderBy(lessons.order);
        
        return {
          ...course,
          lessons: courseLessons
        };
      }));
      
      return {
        ...year,
        courses: coursesWithLessons
      };
    }));


    // Log de auditoría
    await logger.audit({
      userId: user.id,
      action: 'VIEW_CURRICULUM',
      resource: 'curriculum',
      details: { yearCount: years.length }
    });


    return NextResponse.json(curriculum);
  } catch (error) {
    console.error('Error fetching curriculum:', error);
    return NextResponse.json(
      { error: 'Error al obtener el currículo' },
      { status: 500 }
    );
  }
}
```


### **B. Sistema de Notificaciones**


```typescript
// src/app/components/Notifications.tsx


import { useEffect, useState } from 'react';
import { useNotifications } from '@/hooks/useNotifications';
import { useAudio } from '@/hooks/useAudio';


interface Notification {
  id: number;
  title: string;
  message: string;
  type: string;
  read: boolean;
  data?: any;
  createdAt: string;
}


export function Notifications() {
  const [notifications, setNotifications] = useState<Notification[]>([]);
  const [permission, setPermission] = useState<NotificationPermission>('default');
  const { playSound } = useAudio();


  // Solicitar permiso de notificaciones
  useEffect(() => {
    if ('Notification' in window) {
      setPermission(Notification.permission);
    }
  }, []);


  const requestPermission = async () => {
    if ('Notification' in window) {
      const perm = await Notification.requestPermission();
      setPermission(perm);
      if (perm === 'granted') {
        // Registrar Service Worker
        if ('serviceWorker' in navigator) {
          navigator.serviceWorker.register('/sw.js');
        }
      }
    }
  };


  // Mostrar notificación
  const showNotification = (notification: Notification) => {
    // Sonido
    playSound('/notification.mp3');


    // Notificación del navegador
    if ('Notification' in window && Notification.permission === 'granted') {
      new Notification(notification.title, {
        body: notification.message,
        icon: '/icon-192.png',
        badge: '/badge.png',
        vibrate: [200, 100, 200],
        data: notification.data,
      });
    }


    // Agregar al estado
    setNotifications(prev => [notification, ...prev]);


    // Enviar a la API para guardar
    fetch('/api/notifications', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(notification),
    });
  };


  const markAsRead = async (id: number) => {
    await fetch(`/api/notifications/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ read: true }),
    });


    setNotifications(prev =>
      prev.map(n => n.id === id ? { ...n, read: true } : n)
    );
  };


  return (
    <div className="notifications-container">
      {permission !== 'granted' && (
        <button onClick={requestPermission}>
          Activar Notificaciones
        </button>
      )}
      
      <div className="notifications-list">
        {notifications.map((notif) => (
          <div key={notif.id} className={`notification ${notif.read ? 'read' : 'unread'}`}>
            <h3>{notif.title}</h3>
            <p>{notif.message}</p>
            <span className="timestamp">
              {new Date(notif.createdAt).toLocaleString()}
            </span>
            {!notif.read && (
              <button onClick={() => markAsRead(notif.id)}>
                Marcar como leída
              </button>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
```

```javascript
// public/sw.js


self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open('v1').then((cache) => {
      return cache.addAll([
        '/',
        '/manifest.json',
        '/icon-192.png',
        '/notification.mp3'
      ]);
    })
  );
  self.skipWaiting();
});


self.addEventListener('activate', (event) => {
  event.waitUntil(clients.claim());
});


self.addEventListener('push', (event) => {
  const data = event.data?.json() || {};
  
  const options = {
    body: data.body || 'Tarea completada',
    icon: '/icon-192.png',
    badge: '/badge.png',
    vibrate: [200, 100, 200],
    data: data.data || {},
    actions: [
      { action: 'view', title: 'Ver' },
      { action: 'dismiss', title: 'Cerrar' }
    ],
    tag: data.id || 'notification',
    renotify: true,
  };


  event.waitUntil(
    self.registration.showNotification(
      data.title || 'Secure T',
      options
    )
  );
});


// Manejar clicks en notificaciones
self.addEventListener('notificationclick', (event) => {
  event.notification.close();
  
  if (event.action === 'view') {
    const url = event.notification.data.url || '/';
    event.waitUntil(
      clients.openWindow(url)
    );
  }
```

```typescript
    'edit_lessons',
    'review_evidence',
    'view_users',
    'edit_users',
    'view_audit_logs',
    'view_notifications',
    'send_notifications',
  ],
  superadmin: [
    'view_curriculum',
    'edit_curriculum',
    'view_courses',
    'edit_courses',
    'view_lessons',
    'edit_lessons',
    'submit_evidence',
    'review_evidence',
    'view_users',
    'edit_users',
    'manage_roles',
    'view_audit_logs',
    'manage_ai_config',
    'view_notifications',
    'send_notifications',
  ],
};


export function hasPermission(userRole: UserRole, permission: Permission): boolean {
  return rolePermissions[userRole]?.includes(permission) || false;
}


// Middleware de autorización
export async function authorize(user: any, permission: Permission): Promise<boolean> {
  if (!user) return false;
  return hasPermission(user.role as UserRole, permission);
}
```


### **E. Sistema de Auditoría**


```typescript
// src/lib/logger.ts


import { db } from './db';
import { auditLogs } from './db/schema';


interface AuditLogData {
  userId: number;
  action: string;
  resource?: string;
  details?: any;
  ip?: string;
  userAgent?: string;
}


export const logger = {
  async audit(data: AuditLogData) {
    try {
      await db.insert(auditLogs).values({
        userId: data.userId,
        action: data.action,
        resource: data.resource,
        details: data.details || {},
        ip: data.ip,
        userAgent: data.userAgent,
        createdAt: new Date(),
      });
    } catch (error) {
      console.error('Error writing audit log:', error);
    }
  },


  async getAuditLogs(userId?: number, limit = 100) {
    try {
      let query = db.select().from(auditLogs);
      if (userId) {
        query = query.where({ userId });
      }
      query = query.limit(limit).orderBy(auditLogs.createdAt, 'desc');
      return await query;
    } catch (error) {
      console.error('Error fetching audit logs:', error);
      return [];
    }
  }
};
```


### **F. Gobernanza IA**


```typescript
// src/lib/governance.ts


import { db } from './db';
import { aiGovernance } from './db/schema';


interface AIRequest {
  userId: number;
  modelName: string;
  query: string;
  context: any;
}


export class AIGovernance {
  // Validar si una solicitud está permitida
  async validateRequest(request: AIRequest): Promise<{ allowed: boolean; reason?: string }> {
    const config = await this.getConfig(request.modelName);
    
    if (!config) {
      return { allowed: false, reason: 'Modelo no configurado' };
    }


    // Verificar restricciones
    for (const restriction of config.restrictions || []) {
      if (this.matchesRestriction(request, restriction)) {
        return { allowed: false, reason: `Restricción: ${restriction.reason}` };
      }
    }


    // Verificar límites de uso
    const usage = await this.getUsageStats(request.userId, request.modelName);
    if (usage >= (config.maxDailyRequests || 100)) {
      return { allowed: false, reason: 'Límite diario de solicitudes excedido' };
    }


    return { allowed: true };
  }


  async getConfig(modelName: string) {
    const configs = await db.select()
      .from(aiGovernance)
      .where({ modelName })
      .orderBy('version', 'desc')
      .limit(1);
    
    return configs[0];
  }


  async getUsageStats(userId: number, modelName: string) {
    // Implementar conteo de uso
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    
    // Obtener solicitudes del día
    const count = await db
      .select({ count: sql`count(*)` })
      .from(aiAuditLogs)
      .where({
        userId,
        modelName,
        createdAt: sql`>= ${today}`
      });
    
    return count[0]?.count || 0;
  }


  // Registrar uso de IA para auditoría
  async logAIUse(data: {
    userId: number;
    modelName: string;
    query: string;
    response: string;
    duration: number;
    cost: number;
  }) {
    await db.insert(aiAuditLogs).values({
      ...data,
      timestamp: new Date(),
    });


    // Notificar si se excede el umbral
    if (data.cost > 100) {
      await this.sendAlert('AI Cost Alert', `Alto costo: ${data.cost}`);
    }
  }


  matchesRestriction(request: AIRequest, restriction: any): boolean {
    // Implementar lógica de matching
    if (restriction.type === 'content') {
      return request.query.includes(restriction.pattern);
    }
    return false;
  }


  async sendAlert(title: string, message: string) {
    // Enviar notificación administrativa
    const admins = await db.select()
      .from(users)
      .where({ role: 'admin' });
    
    for (const admin of admins) {
      await this.sendNotification(admin.id, title, message);
    }
  }
}
```

```typescript
import { db } from '@/lib/db';
import { academicYears, courses, lessons, competencies } from '@/lib/db/schema';
import { eq, and, asc, sql } from 'drizzle-orm';
import { cache } from 'react';


export class CurriculumService {
  private static instance: CurriculumService;
  private cache: Map<string, any> = new Map();


  static getInstance(): CurriculumService {
    if (!CurriculumService.instance) {
      CurriculumService.instance = new CurriculumService();
    }
    return CurriculumService.instance;

  // Obtener currículo completo con caché
  async getFullCurriculum(forceRefresh = false) {
    const cacheKey = 'full_curriculum';
    
    if (!forceRefresh && this.cache.has(cacheKey)) {
      return this.cache.get(cacheKey);
    }


    try {
      const years = await db.select().from(academicYears).where(eq(academicYears.isActive, true));
      
      const result = await Promise.all(years.map(async (year) => {
        const yearCourses = await db.select()
          .from(courses)
          .where(and(
            eq(courses.academicYearId, year.id),
            eq(courses.isActive, true)
          ))
          .orderBy(asc(courses.order));


        const coursesWithData = await Promise.all(yearCourses.map(async (course) => {
          const [courseLessons, courseCompetencies] = await Promise.all([
            db.select()
              .from(lessons)
              .where(and(
                eq(lessons.courseId, course.id),
                eq(lessons.isPublished, true)
              ))
              .orderBy(asc(lessons.order)),
            db.select()
              .from(competencies)
              .innerJoin(
                courseCompetencies,
                eq(competencies.id, courseCompetencies.competencyId)
              )
              .where(eq(courseCompetencies.courseId, course.id))
          ]);


          return {
            ...course,
            lessons: courseLessons,
            competencies: courseCompetencies,
            totalLessons: courseLessons.length,
            totalHours: courseLessons.reduce((acc, l) => acc + (l.duration || 0), 0),
          };
        }));


        return {
          ...year,
          courses: coursesWithData,
          totalCourses: yearCourses.length,
          totalCredits: yearCourses.reduce((acc, c) => acc + (c.credits || 0), 0),
        };
      }));


      this.cache.set(cacheKey, result);
      return result;
    } catch (error) {
      console.error('Error fetching curriculum:', error);
      throw new Error('Error al obtener el currículo');
    }
  }


  // Obtener un curso específico con todos sus detalles
  async getCourseById(courseId: number) {
    const cacheKey = `course_${courseId}`;
    
    if (this.cache.has(cacheKey)) {
      return this.cache.get(cacheKey);
    }


    try {
      const [course] = await db.select()
        .from(courses)
        .where(eq(courses.id, courseId))
        .limit(1);


      if (!course) {
        throw new Error('Curso no encontrado');
      }


      const [lessons, courseCompetencies, progressStats] = await Promise.all([
        db.select()
          .from(lessons)
          .where(and(
            eq(lessons.courseId, courseId),
            eq(lessons.isPublished, true)
          ))
          .orderBy(asc(lessons.order)),
        db.select()
          .from(competencies)
          .innerJoin(
            courseCompetencies,
            eq(competencies.id, courseCompetencies.competencyId)
          )
          .where(eq(courseCompetencies.courseId, courseId)),
        this.getCourseProgressStats(courseId)
      ]);


      const result = {
        ...course,
        lessons,
        competencies: courseCompetencies,
        stats: progressStats,
      };


      this.cache.set(cacheKey, result);
      return result;
    } catch (error) {
      console.error(`Error fetching course ${courseId}:`, error);
      throw error;
    }
  }


  // Obtener estadísticas de progreso de un curso
  async getCourseProgressStats(courseId: number) {
    const stats = await db
      .select({
        totalStudents: sql<number>`COUNT(DISTINCT ${studentProgress.userId})`,
        completedLessons: sql<number>`SUM(CASE WHEN ${studentProgress.status} = 'completed' THEN 1 ELSE 0 END)`,
        averageScore: sql<number>`AVG(${studentProgress.score})`,
      })
      .from(studentProgress)
      .where(eq(studentProgress.lessonId, courseId));


    return stats[0] || { totalStudents: 0, completedLessons: 0, averageScore: 0 };
  }


  // Buscar cursos por término
  async searchCourses(query: string, filters?: any) {
    const searchTerm = `%${query}%`;
    
    let baseQuery = db.select()
      .from(courses)
      .where(
        sql`${courses.title} ILIKE ${searchTerm} OR ${courses.description} ILIKE ${searchTerm} OR ${courses.code} ILIKE ${searchTerm}`
      );


    if (filters?.category) {
      baseQuery = baseQuery.where(eq(courses.category, filters.category));
    }


    if (filters?.level) {
      baseQuery = baseQuery.where(eq(courses.level, filters.level));
    }


    return await baseQuery.limit(20);
  }


  // Obtener recomendaciones de cursos para un estudiante
  async getRecommendations(userId: number, limit = 5) {
    // Obtener cursos completados
    const completed = await db
      .select({ courseId: courses.id })
      .from(studentProgress)
      .innerJoin(lessons, eq(lessons.id, studentProgress.lessonId))
      .innerJoin(courses, eq(courses.id, lessons.courseId))
      .where(and(
        eq(studentProgress.userId, userId),
        eq(studentProgress.status, 'completed')
      ))
      .groupBy(courses.id);


    const completedIds = completed.map(c => c.courseId);


    // Cursos no completados del mismo nivel
    const recommendations = await db
      .select()
      .from(courses)
      .where(and(
        sql`${courses.id} NOT IN (${completedIds.join(',')})`,
        eq(courses.isActive, true)
      ))
      .orderBy(sql`RANDOM()`)
      .limit(limit);


    return recommendations;
  }


  clearCache() {
    this.cache.clear();
  }
}
```

```typescript
      .from(notifications)
      .where(eq(notifications.userId, userId));


    if (unreadOnly) {
      query = query.where(eq(notifications.read, false));
    }


    query = query
      .orderBy(notifications.createdAt, 'desc')
      .limit(limit)
      .offset(offset);


    const [notifs, unreadCount] = await Promise.all([
      query,
      db.select({ count: sql<number>`COUNT(*)` })
        .from(notifications)
        .where(and(
          eq(notifications.userId, userId),
          eq(notifications.read, false)
        ))
    ]);


    return {
      notifications: notifs,
      unreadCount: unreadCount[0]?.count || 0,
      total: notifs.length,
    };
  }

      .set({

        readAt: new Date()

            .where(eq(pushSubscriptions.endpoint, sub.endpoint));

  }
```

```typescript
// src/lib/services/AuthService.ts


import { db } from '@/lib/db';
import { users } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import { cookies } from 'next/headers';
import { UserRole, rolePermissions, Permission } from '@/lib/rbac';


export class AuthService {
  private static instance: AuthService;
  private readonly JWT_SECRET = process.env.JWT_SECRET!;
  private readonly SALT_ROUNDS = 12;


  static getInstance(): AuthService {
    if (!AuthService.instance) {
      AuthService.instance = new AuthService();
    }
    return AuthService.instance;
  }


  // Registrar usuario
  async register(data: {
    email: string;
    username: string;
    password: string;
    name: string;
    lastName?: string;
  }) {
    // Verificar email único
    const existing = await db.select()
      .from(users)
      .where(eq(users.email, data.email))
      .limit(1);


    if (existing.length > 0) {
      throw new Error('El email ya está registrado');
    }


    // Verificar username único
    const existingUsername = await db.select()
      .from(users)
      .where(eq(users.username, data.username))
      .limit(1);


    if (existingUsername.length > 0) {
      throw new Error('El nombre de usuario ya está en uso');
    }


    // Hash de contraseña
    const passwordHash = await bcrypt.hash(data.password, this.SALT_ROUNDS);


    const [user] = await db.insert(users).values({
      ...data,
      passwordHash,
      role: 'student',
    }).returning();


    // Crear perfil de estudiante
    await db.insert(studentProfiles).values({
      userId: user.id,
    });


    // Generar token
    const token = this.generateToken(user);


    return {
      user: this.sanitizeUser(user),
      token,
    };
  }


  // Login
  async login(email: string, password: string) {
    const [user] = await db.select()
      .from(users)
      .where(eq(users.email, email))
      .limit(1);


    if (!user) {
      throw new Error('Credenciales inválidas');
    }


    const validPassword = await bcrypt.compare(password, user.passwordHash);
    if (!validPassword) {
      throw new Error('Credenciales inválidas');
    }


    // Actualizar último login
    await db.update(users)
      .set({ lastLogin: new Date() })
      .where(eq(users.id, user.id));


    const token = this.generateToken(user);


    return {
      user: this.sanitizeUser(user),
      token,
    };
  }


  // Generar token JWT
  generateToken(user: any) {
    return jwt.sign(
      {
        id: user.id,
        email: user.email,
        role: user.role,
        name: user.name,
      },
      this.JWT_SECRET,
      { expiresIn: '7d' }
    );
  }


  // Verificar token
  async verifyToken(token: string) {
    try {
      const decoded = jwt.verify(token, this.JWT_SECRET) as any;
      
      const [user] = await db.select()
        .from(users)
        .where(eq(users.id, decoded.id))
        .limit(1);


      if (!user || !user.isActive) {
        return null;
      }


      return user;
    } catch (error) {
      return null;
    }
  }


  // Verificar permisos
  async hasPermission(userId: number, permission: Permission) {
    const [user] = await db.select()
      .from(users)
      .where(eq(users.id, userId))
      .limit(1);


    if (!user) return false;
    return rolePermissions[user.role as UserRole]?.includes(permission) || false;
  }


  // Cambiar rol (solo admin)
  async changeRole(userId: number, newRole: UserRole, adminId: number) {
    const hasPermission = await this.hasPermission(adminId, 'manage_roles');
    if (!hasPermission) {
      throw new Error('No tienes permisos para cambiar roles');
    }


    const [user] = await db.update(users)
      .set({ role: newRole })
      .where(eq(users.id, userId))
      .returning();


    return this.sanitizeUser(user);
  }


  // Sanitizar usuario (eliminar datos sensibles)
  sanitizeUser(user: any) {
    const { passwordHash, ...sanitized } = user;
    return sanitized;
  }


  // Refresh token
  async refreshToken(token: string) {
    const user = await this.verifyToken(token);
    if (!user) {
      throw new Error('Token inválido');
    }


    return this.generateToken(user);
  }


  // Logout
  async logout() {
    const cookieStore = cookies();
    cookieStore.delete('token');
  }


  // Middleware para Next.js
  async authMiddleware(req: Request) {
    const token = req.headers.get('Authorization')?.replace('Bearer ', '');
    
    if (!token) {
      return null;
    }


    return await this.verifyToken(token);
  }
}
```

```typescript
    const ip = req.ip || 'unknown';

      return NextResponse.json({ error: 'Demasiadas solicitudes' }, { status: 429 });

    // Obtener parámetros

    const search = url.searchParams.get('search');

    const forceRefresh = url.searchParams.get('refresh') === 'true';


    let data;

    } else if (search) {
      data = await curriculumService.searchCourses(search);
    } else {
      data = await curriculumService.getFullCurriculum(forceRefresh);

    // Log de auditoría
    await logger.audit({
      userId: user.id,
      action: 'VIEW_CURRICULUM',
      resource: courseId ? `course_${courseId}` : 'curriculum',
      details: { 
        search: search || null,
        courseId: courseId || null,
      },
    });


    return NextResponse.json({
      success: true,
      data,
      timestamp: new Date().toISOString(),
    });
  } catch (error: any) {
```

```typescript
    const url = new URL(req.url);
    const lessonId = url.searchParams.get('lessonId');


    let query = db.select()
      .from(studentProgress)
      .where(eq(studentProgress.userId, user.id));


    if (lessonId) {
      query = query.where(eq(studentProgress.lessonId, parseInt(lessonId)));
    }


    const data = await query;


    return NextResponse.json({
      success: true,
      data,
    });
  } catch (error: any) {
    return NextResponse.json({
      success: false,
      error: error.message,
    }, { status: 500 });
  }
}


export async function POST(req: NextRequest) {
  try {
    const user = await authService.authMiddleware(req);
    if (!user) {
      return NextResponse.json({ error: 'No autorizado' }, { status: 401 });
    }


    const body = await req.json();
    const { lessonId, status, progress, score } = body;


    // Verificar que la lección existe
    const [lesson] = await db.select()
      .from(lessons)
      .where(eq(lessons.id, lessonId))
      .limit(1);


    if (!lesson) {
      return NextResponse.json({ error: 'Lección no encontrada' }, { status: 404 });
    }


    // Actualizar o insertar progreso
    const [existing] = await db.select()
      .from(studentProgress)
      .where(and(
        eq(studentProgress.userId, user.id),
        eq(studentProgress.lessonId, lessonId)
      ))
      .limit(1);


    let result;
    if (existing) {
      [result] = await db.update(studentProgress)
        .set({
          status: status || existing.status,
          progress: progress || existing.progress,
          score: score || existing.score,
          updatedAt: new Date(),
          completedAt: status === 'completed' ? new Date() : existing.completedAt,
        })
        .where(eq(studentProgress.id, existing.id))
        .returning();
    } else {
      [result] = await db.insert(studentProgress)
        .values({
          userId: user.id,
          lessonId,
          status: status || 'not_started',
          progress: progress || 0,
          score: score || null,
        })
        .returning();
    }


    // Notificar si se completó
    if (status === 'completed' && !existing?.status !== 'completed') {
      await notificationService.taskCompleted(user.id, {
        taskId: lessonId,
        taskName: lesson.title,
        score: score || 100,
      });
    }


    return NextResponse.json({
      success: true,
      data: result,
    });
  } catch (error: any) {
    return NextResponse.json({
      success: false,
      error: error.message,
    }, { status: 500 });
  }
}
```

```typescript
// src/lib/rateLimit.ts


import { redis } from './redis';


export async function rateLimit(
  key: string,
  prefix: string,
  limit: number,
  window: number
): Promise<{ allowed: boolean; remaining: number; reset: number }> {
  const redisKey = `ratelimit:${prefix}:${key}`;
  const now = Date.now();
  const windowStart = now - window * 1000;


  // Limpiar solicitudes antiguas
  await redis.zremrangebyscore(redisKey, 0, windowStart);


  // Contar solicitudes actuales
  const count = await redis.zcard(redisKey);


  if (count >= limit) {
    return {
      allowed: false,
      remaining: 0,
      reset: Math.floor(now / 1000) + window,
    };
  }


  // Agregar nueva solicitud
  await redis.zadd(redisKey, now, `${now}:${Math.random()}`);
  await redis.expire(redisKey, window);


  return {
    allowed: true,
    remaining: limit - count - 1,
    reset: Math.floor(now / 1000) + window,
  };
}
```

```typescript
import { useState, useEffect, useCallback } from 'react';
import { useAuth } from './useAuth';
import { useWebSocket } from './useWebSocket';


interface Notification {
  id: number;
  title: string;
  message: string;
  type: string;
  read: boolean;
  data?: any;
  createdAt: string;
}


export function useNotifications() {
  const [notifications, setNotifications] = useState<Notification[]>([]);
  const [unreadCount, setUnreadCount] = useState(0);
  const [isLoading, setIsLoading] = useState(false);
  const [permission, setPermission] = useState<NotificationPermission>('default');
  const { user } = useAuth();


  // Solicitar permisos
  const requestPermission = useCallback(async () => {
    if (!('Notification' in window)) {
      console.log('Este navegador no soporta notificaciones');
      return;
    }


    const perm = await Notification.requestPermission();
    setPermission(perm);


    if (perm === 'granted') {
      // Registrar service worker
      if ('serviceWorker' in navigator) {
        try {
          const registration = await navigator.serviceWorker.register('/sw.js');
          console.log('Service Worker registrado:', registration);
        } catch (error) {
          console.error('Error registrando Service Worker:', error);
        }
      }
    }
  }, []);


  // Cargar notificaciones
  const loadNotifications = useCallback(async (unreadOnly = false) => {
    if (!user) return;


    setIsLoading(true);
    try {
      const url = `/api/notifications?${unreadOnly ? 'unread=true' : ''}`;
      const response = await fetch(url);
      const data = await response.json();


      if (data.success) {
        setNotifications(data.data.notifications);
        setUnreadCount(data.data.unreadCount);
      }
    } catch (error) {
      console.error('Error cargando notificaciones:', error);
    } finally {
      setIsLoading(false);
    }
  }, [user]);


  // Marcar como leída
  const markAsRead = useCallback(async (id: number) => {
    if (!user) return;


    try {
      const response = await fetch(`/api/notifications/${id}`, {
        method: 'PATCH',
      });
      const data = await response.json();


      if (data.success) {
        setNotifications(prev =>
          prev.map(n => n.id === id ? { ...n, read: true } : n)
        );
        setUnreadCount(prev => Math.max(0, prev - 1));
      }
    } catch (error) {
      console.error('Error marcando como leída:', error);
    }
  }, [user]);


  // Marcar todas como leídas
  const markAllAsRead = useCallback(async () => {
    if (!user) return;


    try {
      const response = await fetch('/api/notifications/all', {
        method: 'PATCH',
      });
      const data = await response.json();


      if (data.success) {
        setNotifications(prev =>
          prev.map(n => ({ ...n, read: true }))
        );
        setUnreadCount(0);
      }
    } catch (error) {
      console.error('Error marcando todas como leídas:', error);
    }
  }, [user]);


  // Mostrar notificación del navegador
  const showBrowserNotification = useCallback((notification: Notification) => {
    if (permission !== 'granted') return;


    try {
      const navNotification = new Notification(notification.title, {
        body: notification.message,
        icon: '/icon-192.png',
        badge: '/badge.png',
        vibrate: [200, 100, 200],
        data: notification.data,
        tag: `notification-${notification.id}`,
        requireInteraction: true,
      });


      // Sonido
      const audio = new Audio('/notification.mp3');
      audio.play().catch(console.error);


      // Evento click
      navNotification.onclick = () => {
        window.focus();
        navNotification.close();
        if (notification.data?.url) {
          window.location.href = notification.data.url;
        }
      };
    } catch (error) {
      console.error('Error mostrando notificación:', error);
    }
  }, [permission]);


  // WebSocket para notificaciones en tiempo real
  const { sendMessage, lastMessage } = useWebSocket(
    process.env.NEXT_PUBLIC_WS_URL || 'ws://localhost:3000'
  );


  useEffect(() => {
    if (lastMessage) {
      try {
        const data = JSON.parse(lastMessage);
        if (data.type === 'notification') {
          const notification = data.payload;
          setNotifications(prev => [notification, ...prev]);
          setUnreadCount(prev => prev + 1);
          
          if (document.hidden) {
            showBrowserNotification(notification);
          }
        }
      } catch (error) {
        console.error('Error procesando mensaje WebSocket:', error);
      }
    }
  }, [lastMessage, showBrowserNotification]);


  // Cargar notificaciones al inicio
  useEffect(() => {
    if (user) {
      loadNotifications();
    }
  }, [user, loadNotifications]);


  return {
    notifications,
    unreadCount,
    isLoading,
    permission,
    requestPermission,
    loadNotifications,
    markAsRead,
    markAllAsRead,
    showBrowserNotification,
  };
}
```

```typescript
import { useState, useCallback, useEffect } from 'react';


interface AudioOptions {
  volume?: number;
  loop?: boolean;
  autoplay?: boolean;
}


export function useAudio(url: string, options: AudioOptions = {}) {
  const [audio, setAudio] = useState<HTMLAudioElement | null>(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [isLoaded, setIsLoaded] = useState(false);


  // Inicializar audio
  useEffect(() => {
    if (typeof window === 'undefined') return;


    const audioElement = new Audio(url);
    audioElement.volume = options.volume || 0.5;
    audioElement.loop = options.loop || false;
    
    audioElement.addEventListener('canplaythrough', () => {
      setIsLoaded(true);
    });


    audioElement.addEventListener('ended', () => {
      setIsPlaying(false);
    });


    setAudio(audioElement);


    // Autoplay
    if (options.autoplay) {
      play();
    }


    return () => {
      audioElement.pause();
      audioElement.removeEventListener('ended', () => {});
    };
  }, [url]);


  // Reproducir
  const play = useCallback(async () => {
    if (!audio) return;


    try {
      await audio.play();
      setIsPlaying(true);
    } catch (error) {
      console.error('Error reproduciendo audio:', error);
    }
  }, [audio]);


  // Pausar
  const pause = useCallback(() => {
    if (!audio) return;
    audio.pause();
    setIsPlaying(false);
  }, [audio]);


  // Reproducir sonido de notificación
  const playNotification = useCallback(async (type: 'task' | 'message' | 'alert' = 'task') => {
    const sounds = {
      task: '/sounds/achievement.mp3',
      message: '/sounds/message.mp3',
      alert: '/sounds/alert.mp3',
    };


    const soundUrl = sounds[type] || sounds.task;
    const sound = new Audio(soundUrl);
    sound.volume = 0.7;
    
    try {
      await sound.play();
    } catch (error) {
      console.error('Error reproduciendo sonido de notificación:', error);
    }
  }, []);


  // Cambiar volumen
  const setVolume = useCallback((volume: number) => {
    if (!audio) return;
    const clamped = Math.max(0, Math.min(1, volume));
    audio.volume = clamped;
  }, [audio]);


  return {
    play,
    pause,
    isPlaying,
    isLoaded,
    playNotification,
    setVolume,
    audio,
  };
}
```

### **C. Hook de WebSocket**


```typescript
// src/hooks/useWebSocket.ts


import { useState, useEffect, useCallback, useRef } from 'react';


interface WebSocketOptions {
  reconnectInterval?: number;
  maxRetries?: number;
  onOpen?: () => void;
  onClose?: () => void;
  onError?: (error: Event) => void;
}


export function useWebSocket(url: string, options: WebSocketOptions = {}) {
  const {
    reconnectInterval = 3000,
    maxRetries = 5,
    onOpen,
    onClose,
    onError,
  } = options;


  const [isConnected, setIsConnected] = useState(false);
  const [lastMessage, setLastMessage] = useState<string | null>(null);
  const [retryCount, setRetryCount] = useState(0);


  const wsRef = useRef<WebSocket | null>(null);
  const reconnectTimeoutRef = useRef<NodeJS.Timeout | null>(null);


  // Conectar
  const connect = useCallback(() => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      return;
    }


    try {
      const ws = new WebSocket(url);
      wsRef.current = ws;


      ws.onopen = () => {
        setIsConnected(true);
        setRetryCount(0);
        onOpen?.();
        console.log('WebSocket conectado');
      };


      ws.onclose = () => {
        setIsConnected(false);
        onClose?.();
        console.log('WebSocket desconectado');


        // Reconectar
        if (retryCount < maxRetries) {
          reconnectTimeoutRef.current = setTimeout(() => {
            setRetryCount(prev => prev + 1);
            connect();
          }, reconnectInterval);
        }
      };


      ws.onerror = (error) => {
        console.error('WebSocket error:', error);
        onError?.(error);
      };


      ws.onmessage = (event) => {
        setLastMessage(event.data);
      };
    } catch (error) {
      console.error('Error creando WebSocket:', error);
    }
  }, [url, retryCount, maxRetries, reconnectInterval, onOpen, onClose, onError]);


  // Desconectar
  const disconnect = useCallback(() => {
    if (reconnectTimeoutRef.current) {
      clearTimeout(reconnectTimeoutRef.current);
      reconnectTimeoutRef.current = null;
    }


    if (wsRef.current) {
      wsRef.current.close();
      wsRef.current = null;
      setIsConnected(false);
    }
  }, []);


  // Enviar mensaje
  const sendMessage = useCallback((data: any) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(data));
      return true;
    }
    console.warn('WebSocket no está conectado');
    return false;
  }, []);


  // Conectar automáticamente
  useEffect(() => {
    connect();


    return () => {
      disconnect();
    };
  }, [connect, disconnect]);


  return {
    isConnected,
    lastMessage,
    sendMessage,
    connect,
    disconnect,
    ws: wsRef.current,
  };
}
```

```nginx
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;


            # Compresión
            gzip on;
            gzip_vary on;
            gzip_min_length 1024;
            gzip_types text/plain text/css text/xml text/javascript application/javascript application/xml+rss application/json;


            # Caché
            location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf)$ {
                expires 1y;
                add_header Cache-Control "public, immutable";
            }
        }


        location /api {
            proxy_pass http://app;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            limit_req zone=api burst=20 nodelay;
        }


        location /ws {
            proxy_pass http://app;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
        }


        # Health check
        location /health {
            proxy_pass http://app;
            access_log off;
        }
    }
```

```typescript
// src/lib/monitoring.ts


import winston from 'winston';
import { ElasticsearchTransport } from 'winston-elasticsearch';


const logger = winston.createLogger({
  level: process.env.LOG_LEVEL || 'info',
  format: winston.format.combine(
    winston.format.timestamp(),
    winston.format.json()
  ),
  transports: [
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.colorize(),
        winston.format.simple()
      )
    }),
    new winston.transports.File({ 
      filename: 'logs/error.log', 
      level: 'error' 
    }),
    new winston.transports.File({ 
      filename: 'logs/combined.log' 
    }),
  ],
});


// Integración con Elasticsearch
if (process.env.ELASTICSEARCH_URL) {
  logger.add(new ElasticsearchTransport({
    level: 'info',
    clientOpts: {
      node: process.env.ELASTICSEARCH_URL,
      auth: {
        username: process.env.ELASTICSEARCH_USER || '',
        password: process.env.ELASTICSEARCH_PASSWORD || '',
      },
    },
    index: 'secure-t-logs',
  }));
}


export function logRequest(req: Request, res: Response, duration: number) {
  logger.info('Request', {
    method: req.method,
    url: req.url,
    status: res.status,
    duration,
    userAgent: req.headers.get('user-agent'),
    ip: req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip'),
  });
}


export function logError(error: Error, context?: any) {
  logger.error('Error', {
    error: error.message,
    stack: error.stack,
    context,
  });
}


export const metrics = {
  increment: (metric: string, value = 1) => {
    logger.info('Metric', { metric, value, timestamp: Date.now() });
  },
  timing: (metric: string, duration: number) => {
    logger.info('Timing', { metric, duration, timestamp: Date.now() });
  },
};
```
